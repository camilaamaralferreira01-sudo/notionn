function dateTime() { var today = new Date(); // Dia da semana + data em português var options = { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' }; var dataFormatada = today.toLocaleDateString('pt-BR', options); // Hora formatada em português (24h) var horaFormatada = today.toLocaleTimeString('pt-BR'); // Exibe tudo junto document.getElementById('date').innerHTML = dataFormatada + " - " + horaFormatada; // Saudação var hrs = today.getHours(); var greet; if (hrs < 12) greet = 'Bom dia'; else if (hrs >= 12 && hrs <= 17) greet = 'Boa tarde'; else greet = 'Boa noite'; document.getElementById('greet').innerHTML = greet; setTimeout(dateTime, 1000); } dateTime();
